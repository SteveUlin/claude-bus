cmake_minimum_required(VERSION 3.25)
project(claude-bus CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Emit compile_commands.json so clangd/LSP can resolve C++23 headers
# (std::print, std::expected, ...). A symlink at the repo root points
# at build/compile_commands.json; see .clangd for additional flags.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_compile_options(-Wall -Wextra -Wpedantic)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)

# Pure, dependency-light broker logic — no zellij, no sockets, no clock
# beyond what the caller passes. Linked into both the `bus` binary and
# the `bus_tests` unit suite so the tested code IS the shipped code.
# (tail_reader is header-only; computeAxes lives in bus_readers.)
add_library(bus_core STATIC
    src/event.cpp
    src/json_min.cpp
    src/recovery.cpp
    src/journal.cpp
    src/cursor_store.cpp
    src/peer_registry.cpp
)
# Envelope codec — messaging layer above the domain-agnostic Journal kernel.
# bus_core must NOT depend on it; only the delivery binary and bus_readers
# (which decode envelope fields for task_model etc.) link it.
add_library(bus_envelope STATIC
    src/envelope.cpp
)
target_include_directories(bus_core PUBLIC src)

# The Readers seam (broker-seam-redesign §2/§6): pure derivations over
# events.jsonl + an INJECTED PaneState value. Links bus_core ONLY — no pane
# acquisition, no sockets, no clock beyond the caller's. That link boundary is
# the structural half of the no-authority guard: a reader that grows a mutable
# field or reaches for pane.h fails to build here, broker-free, which is the §3
# litmus passed by construction. The TUs stay separate free-function units —
# the "Readers" component is conceptual, not one class. (The verify reader is
# the one-shot `bus verify` in src/sub/, still bound to its CLI shell.)
add_library(bus_readers STATIC
    src/agent_status.cpp
    src/context_stats.cpp
    src/task_model.cpp
    src/trigger_feed.cpp
)
target_include_directories(bus_readers PUBLIC src)
# readAgents / deriveTrig parse each line with json::parse (json_min), so the
# json symbols must resolve after this archive on the link line.
target_link_libraries(bus_readers PUBLIC bus_core bus_envelope)

# Child-process spawn: fork/exec + stdout-capture + timeout-kill (process.h).
# One implementation of the machinery that was hand-rolled twice (pane.cpp's
# runCapture/runSilent and sub_lifecycle.cpp's runSync/runCaptureLocal). No deps
# beyond libc; testable in isolation (tests/unit/test_process.cpp).
add_library(bus_process STATIC
    src/process.cpp
)
target_include_directories(bus_process PUBLIC src)

# Pure parser for zellij `dump-screen --ansi` (pane_parse.h): string-in /
# struct-out mode/bypass/input detection, no syscalls. Links NOTHING — that
# empty dep set is the structural guard: a parser that reaches for a pane
# actuator or subprocess fails to build here. Unit-tested in isolation
# (tests/unit/test_pane_parse.cpp); pane.cpp's paneState is the production caller.
add_library(bus_pane_parse STATIC
    src/pane_parse.cpp
)
target_include_directories(bus_pane_parse PUBLIC src)

# Pane acquisition + actuators (paneState, sendToPaneSafe, the loop-thread TTL
# cache). The loop/Daemon plane: single-thread-owned mutable caches that must
# NOT enter Readers. Kept its own unit so the read plane can't pull pane.cpp by
# accident — bus_readers does not link it, only the binary + tests do.
add_library(bus_pane STATIC
    src/pane.cpp
)
target_include_directories(bus_pane PUBLIC src)
# pane.cpp drives zellij through the shared spawn module and parses dumps via
# the pure parser.
target_link_libraries(bus_pane PUBLIC bus_process bus_pane_parse)

# The Policy seam (docs/policy-actors.md): autonomous broker behaviors as
# actors that observe a Readers snapshot, apply their own guards, and return
# DECLARATIVE actions the loop executes. Links bus_readers + bus_core ONLY — no
# bus_pane, no socket. That link boundary is the structural guard: an actor that
# reaches for pane acquisition or a shared mutable broker field fails to build
# here. RecoveryActor (the reference actor) is the extracted P2 auto-recovery.
add_library(bus_policy STATIC
    src/policy.cpp
    src/recovery_actor.cpp
    src/dispatch_actor.cpp
    src/blackboard_actor.cpp
)
target_include_directories(bus_policy PUBLIC src)
target_link_libraries(bus_policy PUBLIC bus_readers)

# Build-commit stamp (D4 code-drift detection). build_info.h carries the commit
# the binary was built from; agent-launch compares it against landed main and
# auto-rebuilds bin/bus on drift. Colocated git HEAD == the jj commit_id, so it
# matches `jj log -r main`. The stamp is regenerated BY THE BUILD (via the
# stamp_build_info target below), not only at configure, so `bus version` always
# reflects the source actually compiled. The -P script's configure_file is
# copy-if-different, so an unchanged commit rewrites nothing and triggers no
# recompile. "unknown" when git is unavailable.
set(BUS_STAMP_ARGS
    -DSRC_DIR=${CMAKE_SOURCE_DIR}
    -DIN_FILE=${CMAKE_SOURCE_DIR}/src/build_info.h.in
    -DOUT_FILE=${CMAKE_BINARY_DIR}/build_info.h)
# Generate once at configure so the header exists for IDEs and the first compile.
execute_process(COMMAND ${CMAKE_COMMAND} ${BUS_STAMP_ARGS}
                -P ${CMAKE_SOURCE_DIR}/cmake/stamp_build_info.cmake)
# Refresh on every build so the stamp tracks incremental builds, not just configure.
add_custom_target(stamp_build_info ALL
    COMMAND ${CMAKE_COMMAND} ${BUS_STAMP_ARGS}
            -P ${CMAKE_SOURCE_DIR}/cmake/stamp_build_info.cmake
    BYPRODUCTS ${CMAKE_BINARY_DIR}/build_info.h
    COMMENT "Stamping build commit"
    VERBATIM)

# Unit tests for the pure logic (json::parse, bus::Journal::parseForTest,
# computeAxes, the TailReader split/offset dance). Minimal in-tree
# harness, no third-party dependency. Binary lands in build/, not bin/.
# Run with `ctest --test-dir build` or `./build/bus_tests`.
enable_testing()
# bus_tests links bus_core normally. Cross-journal ack now calls fatal()
# (always-on crash) instead of assert() + return false, so no NDEBUG
# re-compilation is needed. The cross_journal_ack_refused test verifies
# the return value on the SAME-journal path; the bad-call crash is tested
# separately via fork-based death tests (Phase 2).
add_executable(bus_tests
    tests/unit/harness.cpp
    tests/unit/test_json.cpp
    tests/unit/test_event.cpp
    tests/unit/test_journal.cpp
    tests/unit/test_cursor_store.cpp
    tests/unit/test_peer_registry.cpp
    tests/unit/test_envelope.cpp
    tests/unit/test_agent_status.cpp
    tests/unit/test_context_stats.cpp
    tests/unit/test_fold.cpp
    tests/unit/test_readiness.cpp
    tests/unit/test_doorbell_readiness.cpp
    tests/unit/test_tail_reader.cpp
    tests/unit/test_pane_cache.cpp
    tests/unit/test_pane_parse.cpp
    tests/unit/test_process.cpp
    tests/unit/test_topic_registry.cpp
    tests/unit/test_task_graph.cpp
    tests/unit/test_recovery.cpp

    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
add_test(NAME unit COMMAND bus_tests)
set_tests_properties(unit PROPERTIES LABELS unit)

# Integration tests: real broker + fake-zellij harness, hermetic ($STATE
# isolated per-test via mktemp, self-cleaning via EXIT trap). Behind the
# `itest` label so `ctest -L unit` stays fast; `ctest -L itest` runs the
# real delivery-loop gate. RUN_SERIAL because each spawns its own broker
# and binds a Unix socket — no concurrency between them.
file(GLOB BUS_ITESTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/*-itest.sh)
foreach(itest_path ${BUS_ITESTS})
  get_filename_component(itest_name ${itest_path} NAME_WE)
  add_test(NAME ${itest_name} COMMAND bash ${itest_path}
           WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  set_tests_properties(${itest_name} PROPERTIES
      LABELS itest
      RUN_SERIAL TRUE
      TIMEOUT 180)
endforeach()
